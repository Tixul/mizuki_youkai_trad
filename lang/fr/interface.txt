# いれかえ = Place(r), ぞくせい = Élément, ようりょく = Yôryoku, れいりょく = Esprit, はやさ = Vitesse,
# ようかいかんど = Flair, さつえいのうりょく = Technique, しょじきん / おかね = Argent.
